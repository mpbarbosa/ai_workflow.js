export function isEdgeCase(similarity) {
  return (
    similarity >= CONFIDENCE_THRESHOLDS.EDGE_CASE_MIN &&
    similarity < CONFIDENCE_THRESHOLDS.EDGE_CASE_MAX
  );
}
